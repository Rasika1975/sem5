EXPERIMENT 01

Leaky Bucket Algorithm

Python
capacity = int(input("Enter bucket capacity: "))
rate = int(input("Enter output rate: "))
n = int(input("Enter number of packets: "))

bucket = 0

for i in range(n):
    packet = int(input("Enter packet size: "))

    if bucket + packet <= capacity:
        bucket += packet
        print("Packet accepted")
    else:
        print("Packet dropped")

    sent = min(bucket, rate)
    bucket -= sent

    print("Data sent:", sent)
    print("Remaining data:", bucket)

print("Transmission completed")

Sliding Window Protocol

    end = min(i + w - 1, n)

EXPERIMENT 05

Distance Vector Routing Algorithm

Python
INF = 999

# Cost between routers
cost = [
    [0,   1,   4, INF],
    [1,   0,   2,   6],
    [4,   2,   0,   3],
    [INF, 6,   3,   0]
]

nodes = ["A", "B", "C", "D"]
dist = [row[:] for row in cost]

# Update routing tables
while True:
    new_dist = [row[:] for row in dist]

    for i in range(4):
        for j in range(4):
            for k in range(4):
                if cost[i][k] != INF:
                    new_dist[i][j] = min(
                        new_dist[i][j],
                        cost[i][k] + dist[k][j]
                    )

    if new_dist == dist:
        break

    dist = new_dist

# Display final routing table
print("Final Distance Table:")

for i in range(4):
    print("Router", nodes[i], end=": ")
    for j in range(4):
        print(nodes[j], "=", dist[i][j], end="  ")
    print()

Additional Distance Vector Routing Code

Python

The original routing-table program above is kept unchanged. This is the additional version you supplied.

# Distance Vector Routing Algorithm

n = int(input("Enter number of routers: "))

print("Enter cost matrix (use 999 for no connection):")

cost = []
for i in range(n):
    row = list(map(int, input().split()))
    cost.append(row)

dist = [row[:] for row in cost]

# Distance Vector Algorithm
for k in range(n):
    for i in range(n):
        for j in range(n):
            if dist[i][k] + dist[k][j] < dist[i][j]:
                dist[i][j] = dist[i][k] + dist[k][j]

print("\nFinal Routing Table:")

for i in range(n):
    print("\nRouter", i + 1)
    for j in range(n):
        print("To Router", j + 1, "Cost =", dist[i][j])
EXPERIMENT 06

Socket Programming — PING

Python

Raw sockets may require administrator/root permissions and may not work in online compilers.

import socket
import struct
import time
import os
import sys

def checksum(data):
    total = 0

    for i in range(0, len(data) - 1, 2):
        total += (data[i] << 8) + data[i + 1]

    if len(data) % 2:
        total += data[-1] << 8

    total = (total >> 16) + (total & 0xffff)
    total += total >> 16

    return ~total & 0xffff


host = input("Enter destination: ").strip()

try:
    address = socket.gethostbyname(host)

    sock = socket.socket(
        socket.AF_INET,
        socket.SOCK_RAW,
        socket.IPPROTO_ICMP
    )

    sock.settimeout(3)

    packet_id = os.getpid() & 0xffff
    data = b"Hello"

    header = struct.pack("!BBHHH", 8, 0, 0, packet_id, 1)
    check = checksum(header + data)
    header = struct.pack(
        "!BBHHH", 8, 0, check, packet_id, 1
    )

    start = time.time()
    sock.sendto(header + data, (address, 1))

    reply, sender = sock.recvfrom(1024)
    elapsed = (time.time() - start) * 1000

    header_length = (reply[0] & 15) * 4
    icmp = reply[header_length:header_length + 8]
    reply_type = struct.unpack("!BBHHH", icmp)[0]

    if reply_type == 0:
        print("Reply received from", sender[0])
        print("Time:", round(elapsed, 2), "ms")
    else:
        print("ICMP reply received, type:", reply_type)

except socket.timeout:
    print("Request timed out.")

except (PermissionError, OSError) as error:
    print("Socket error:", error)
    print("Try running locally with suitable permissions.")

finally:
    if "sock" in locals():
        sock.close()

IP Address Configuration

Python

This program reads the Windows ipconfig output and displays IP details. It does not change network settings.

# IP Address Configuration

import socket
import subprocess

hostname = socket.gethostname()
result = subprocess.getoutput("ipconfig")

print("IP Address Configuration")
print("Host Name:", hostname)

for line in result.splitlines():
    if "IPv4 Address" in line:
        print("IP Address:", line.split(":")[-1].strip())

    elif "Subnet Mask" in line:
        print("Subnet Mask:", line.split(":")[-1].strip())

    elif "Default Gateway" in line:
        gateway = line.split(":")[-1].strip()
        if gateway:
            print("Default Gateway:", gateway)
EXPERIMENT 08

Socket Programming — TRACEROUTE

Python

Raw sockets may require administrator/root permissions and may not work in online compilers.

import os
import socket
import struct
import time

ICMP_ECHO_REQUEST = 8
ICMP_CODE = 0
TIMEOUT = 2.0
MAX_HOPS = 30


def checksum(source_string):
    """Calculate the checksum for the ICMP packet header."""
    sum_val = 0
    count_to = (len(source_string) // 2) * 2
    count = 0
    while count < count_to:
        this_val = source_string[count + 1] * 256 + source_string[count]
        sum_val = sum_val + this_val
        sum_val = sum_val & 0xFFFFFFFF
        count = count + 2
    if count_to < len(source_string):
        sum_val = sum_val + source_string[len(source_string) - 1]
        sum_val = sum_val & 0xFFFFFFFF
    sum_val = (sum_val >> 16) + (sum_val & 0xFFFF)
    sum_val = sum_val + (sum_val >> 16)
    answer = ~sum_val
    answer = answer & 0xFFFF
    answer = answer >> 8 | (answer << 8 & 0xFF00)
    return answer


def create_icmp_packet(packet_id):
    """Create an ICMP echo request packet."""
    header = struct.pack("bbHHh", ICMP_ECHO_REQUEST, ICMP_CODE, 0, packet_id, 1)
    data = struct.pack("d", time.time())
    my_checksum = checksum(header + data)
    header = struct.pack(
        "bbHHh", ICMP_ECHO_REQUEST, ICMP_CODE, socket.htons(my_checksum), packet_id, 1
    )
    return header + data


def traceroute(dest_name):
    dest_addr = socket.gethostbyname(dest_name)
    print(f"Traceroute to {dest_name} ({dest_addr}) with max hops {MAX_HOPS}:\n")
    print("Hop\tTime (ms)\tRouter IP Address")
    print("-" * 40)

    icmp = socket.getprotobyname("icmp")

    for ttl in range(1, MAX_HOPS + 1):
        try:
            my_socket = socket.socket(socket.AF_INET, socket.SOCK_RAW, icmp)
        except PermissionError:
            print(
                "Error: Raw sockets require Administrator or Root privileges. Please"
                " run your script as Administrator."
            )
            return

        # Set the Time-to-Live (TTL) socket option
        my_socket.setsockopt(socket.IPPROTO_IP, socket.IP_TTL, ttl)
        my_socket.settimeout(TIMEOUT)

        packet_id = os.getpid() & 0xFFFF
        packet = create_icmp_packet(packet_id)

        start_time = time.time()
        try:
            my_socket.sendto(packet, (dest_addr, 1))
            _, curr_addr = my_socket.recvfrom(512)
            curr_addr = curr_addr[0]
            elapsed_time = (time.time() - start_time) * 1000

            print(f"{ttl}\t{elapsed_time:.2f} ms\t{curr_addr}")

            # Stop tracing if destination is reached
            if curr_addr == dest_addr:
                print("\nTrace complete.")
                break

        except socket.timeout:
            print(f"{ttl}\t* * *\tRequest timed out.")
        finally:
            my_socket.close()


if __name__ == "__main__":
    # Target destination (e.g., Google Public DNS)
    target = "8.8.8.8"
    traceroute(target)

FTP File Transfer — Server

Java

Run FileServer.java first, then FileClient.java in a separate terminal. The source is arranged from your pasted file.

import java.io.*;
import java.net.*;

public class FileServer {
    public static void main(String[] args) {
        int port = 5000;

        try {
            ServerSocket serverSocket = new ServerSocket(port);
            System.out.println("Server started...");
            System.out.println("Waiting for client...");

            Socket socket = serverSocket.accept();
            System.out.println("Client connected.");

            BufferedReader reader = new BufferedReader(
                new InputStreamReader(socket.getInputStream()));

            String fileName = reader.readLine();
            System.out.println("Requested file: " + fileName);

            File file = new File(fileName);

            PrintWriter writer = new PrintWriter(
                socket.getOutputStream(), true);

            if (!file.exists()) {
                writer.println("ERROR");
                System.out.println("File not found.");
            } else {
                writer.println("OK");

                FileInputStream fileInputStream =
                    new FileInputStream(file);

                OutputStream outputStream =
                    socket.getOutputStream();

                byte[] buffer = new byte[1024];
                int bytesRead;

                while ((bytesRead =
                        fileInputStream.read(buffer)) != -1) {
                    outputStream.write(buffer, 0, bytesRead);
                }

                outputStream.flush();
                fileInputStream.close();

                System.out.println("File sent successfully.");
            }

            socket.close();
            serverSocket.close();

        } catch (Exception e) {
            System.out.println("Error: " + e.getMessage());
        }
    }
}

FTP File Transfer — Client

Java

Run FileServer.java first, then FileClient.java in a separate terminal. The source is arranged from your pasted file.

import java.io.*;
import java.net.*;

public class FileClient {
    public static void main(String[] args) {
        String serverIP = "127.0.0.1";
        int port = 5000;

        try {
            Socket socket = new Socket(serverIP, port);
            System.out.println("Connected to server.");

            BufferedReader keyboard = new BufferedReader(
                new InputStreamReader(System.in));

            System.out.print("Enter file name: ");
            String fileName = keyboard.readLine();

            PrintWriter writer = new PrintWriter(
                socket.getOutputStream(), true);

            writer.println(fileName);

            BufferedReader reader = new BufferedReader(
                new InputStreamReader(socket.getInputStream()));

            String status = reader.readLine();

            if (status.equals("OK")) {
                String outputFile = "received_" + fileName;

                FileOutputStream fileOutputStream =
                    new FileOutputStream(outputFile);

                InputStream inputStream =
                    socket.getInputStream();

                byte[] buffer = new byte[1024];
                int bytesRead;

                while ((bytesRead =
                        inputStream.read(buffer)) != -1) {
                    fileOutputStream.write(
                        buffer, 0, bytesRead);
                }

                fileOutputStream.close();

                System.out.println("File received successfully.");
                System.out.println("Saved as: " + outputFile);

            } else {
                System.out.println("File not found on server.");
            }

            socket.close();

        } catch (Exception e) {
            System.out.println("Error: " + e.getMessage());
        }
    }
}

Python File Transfer — Client

Python

Run the Python server first, then this client in a second terminal. Put the file to send in the client working folder.

# File Transfer Client

import socket
import os

HOST = "127.0.0.1"
PORT = 5001

filename = input("Enter file name to send: ")

if not os.path.isfile(filename):
    print("File not found!")
else:
    client = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    client.connect((HOST, PORT))

    with open(filename, "rb") as file:
        while True:
            data = file.read(1024)
            if not data:
                break
            client.sendall(data)

    client.shutdown(socket.SHUT_WR)

    message = client.recv(1024).decode()
    print("Server:", message)

    client.close()

Python File Transfer — Server

Python

Run this server first. It saves received data as received_file.txt; then run the client in a separate terminal.

# File Transfer Server

import socket

HOST = "127.0.0.1"
PORT = 5001

server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server.bind((HOST, PORT))
server.listen(1)

print("Server is waiting for client...")

conn, addr = server.accept()
print("Connected to:", addr)

with conn:
    with open("received_file.txt", "wb") as file:
        while True:
            data = conn.recv(1024)
            if not data:
                break
            file.write(data)

    conn.sendall(b"File received successfully!")

print("File received successfully.")
server.close()

import subprocess

domain = input("Enter domain name: ")

try:
    result = subprocess.run(
        ["nslookup", domain],
        capture_output=True,

        timeout=10

except FileNotFoundError:
