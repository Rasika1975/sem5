Experiment 3: Water Jug Problem using BFS

Aim: To solve the Water Jug Problem using Breadth-First Search (BFS).

Python Code
# Experiment 3: Water Jug Problem using BFS

from collections import deque

def water_jug(capacity1, capacity2, target):
    queue = deque([(0, 0)])
    visited = set()

    while queue:
        x, y = queue.popleft()

        if (x, y) in visited:
            continue

        visited.add((x, y))
        print("Jug 1:", x, "Jug 2:", y)

        if x == target or y == target:
            print("Target amount reached!")
            return

        states = [
            (capacity1, y),  # Fill Jug 1
            (x, capacity2),  # Fill Jug 2
            (0, y),          # Empty Jug 1
            (x, 0),          # Empty Jug 2
            (max(0, x - (capacity2 - y)),
             min(capacity2, x + y)),  # Pour Jug 1 to Jug 2
            (min(capacity1, x + y),
             max(0, y - (capacity1 - x)))  # Pour Jug 2 to Jug 1
        ]

        for state in states:
            if state not in visited:
                queue.append(state)

    print("Target amount cannot be measured.")

# Jug capacities and target
water_jug(4, 3, 2)

Experiment 3A: N-Queen Problem

Aim: To solve the N-Queen Problem using backtracking.

Python Code
# Experiment 3A: N-Queen Problem

N = 4
board = [-1] * N

def is_safe(row, col):
    for i in range(row):
        if board[i] == col:
            return False

        if abs(board[i] - col) == abs(i - row):
            return False

    return True

def solve(row):
    if row == N:
        return True

    for col in range(N):
        if is_safe(row, col):
            board[row] = col

            if solve(row + 1):
                return True

            board[row] = -1

    return False

if solve(0):
    print("Solution for", N, "Queens:")
    for row in range(N):
        for col in range(N):
            if board[row] == col:
                print("Q", end=" ")
            else:
                print(".", end=" ")
        print()
else:
    print("No solution exists")

Experiment 3B: 8-Puzzle using A* Search

Aim: To solve the 8-Puzzle Problem using the A* search algorithm.

Python Code
# Experiment 3B: 8-Puzzle using A* Search

import heapq

start = (1, 2, 3,
         4, 0, 6,
         7, 5, 8)

goal = (1, 2, 3,
        4, 5, 6,
        7, 8, 0)

def heuristic(state):
    distance = 0

    for i, value in enumerate(state):
        if value != 0:
            target = goal.index(value)
            distance += abs(i // 3 - target // 3)
            distance += abs(i % 3 - target % 3)

    return distance

def get_neighbors(state):
    zero = state.index(0)
    row, col = zero // 3, zero % 3
    neighbors = []

    moves = [(-1, 0), (1, 0),
             (0, -1), (0, 1)]

    for dr, dc in moves:
        r, c = row + dr, col + dc

        if 0 <= r < 3 and 0 <= c < 3:
            new_pos = r * 3 + c
            temp = list(state)

            temp[zero], temp[new_pos] = (
                temp[new_pos], temp[zero]
            )

            neighbors.append(tuple(temp))

    return neighbors

def solve_puzzle(start):
    pq = [(heuristic(start), 0, start)]
    parent = {start: None}
    cost = {start: 0}

    while pq:
        f, g, state = heapq.heappop(pq)

        if state == goal:
            path = []

            while state is not None:
                path.append(state)
                state = parent[state]

            return path[::-1]

        if g != cost[state]:
            continue

        for nxt in get_neighbors(state):
            new_cost = g + 1

            if new_cost < cost.get(nxt, float("inf")):
                cost[nxt] = new_cost
                parent[nxt] = state

                priority = new_cost + heuristic(nxt)
                heapq.heappush(
                    pq, (priority, new_cost, nxt)
                )

    return None

path = solve_puzzle(start)

if path:
    print("Solution found!")
    print("Number of moves:", len(path) - 1)

    for state in path:
        for i in range(0, 9, 3):
            print(state[i:i + 3])
        print()
else:
    print("No solution exists")

Experiment 3C: Travelling Salesman Problem

Aim: To find the minimum-cost route for the Travelling Salesman Problem by checking possible city orders.

Python Code
# Experiment 3C: Travelling Salesman Problem

from itertools import permutations

# Distance between cities
graph = [
    [0, 10, 15, 20],
    [10, 0, 35, 25],
    [15, 35, 0, 30],
    [20, 25, 30, 0]
]

n = len(graph)
start = 0

min_cost = float("inf")
best_path = None

# Try every possible order of remaining cities
for route in permutations(
    [i for i in range(n) if i != start]
):
    path = (start,) + route + (start,)

    cost = 0

    for i in range(len(path) - 1):
        cost += graph[path[i]][path[i + 1]]

    if cost < min_cost:
        min_cost = cost
        best_path = path

print("Best route:")
print(" -> ".join(str(city) for city in best_path))

print("Minimum cost:", min_cost)

Experiment 4A: A* Search Algorithm

Aim: To implement the A* search algorithm to find a path with minimum cost.

Python Code
# Experiment 4A: A* Search Algorithm

graph = {
    'A': [('B', 1), ('C', 4)],
    'B': [('C', 2), ('D', 5)],
    'C': [('D', 1)],
    'D': []
}

heuristic = {
    'A': 4,
    'B': 3,
    'C': 1,
    'D': 0
}

def astar(start, goal):
    open_list = [(heuristic[start], 0, start, [start])]
    visited = set()

    while open_list:
        open_list.sort()
        f, g, node, path = open_list.pop(0)

        if node == goal:
            print("Shortest path:", " -> ".join(path))
            print("Total cost:", g)
            return

        if node in visited:
            continue

        visited.add(node)

        for neighbor, cost in graph[node]:
            if neighbor not in visited:
                new_g = g + cost
                new_f = new_g + heuristic[neighbor]

                open_list.append(
                    (new_f, new_g, neighbor, path + [neighbor])
                )

    print("Path not found")

astar('A', 'D')

Experiment 4B: Alpha-Beta Pruning

Aim: To implement Alpha-Beta Pruning to find the optimal value in a game tree.

Python Code
# Experiment 4B: Alpha-Beta Pruning

def alphabeta(depth, node, maximizing,
              values, alpha, beta):

    if depth == 3:
        return values[node]

    if maximizing:
        best = float('-inf')

        for i in range(2):
            value = alphabeta(
                depth + 1,
                node * 2 + i,
                False,
                values,
                alpha,
                beta
            )

            best = max(best, value)
            alpha = max(alpha, best)

            if alpha >= beta:
                break

        return best

    else:
        best = float('inf')

        for i in range(2):
            value = alphabeta(
                depth + 1,
                node * 2 + i,
                True,
                values,
                alpha,
                beta
            )

            best = min(best, value)
            beta = min(beta, best)

            if alpha >= beta:
                break

        return best


# Leaf-node values of a binary game tree
values = [3, 5, 6, 9, 1, 2, 0, -1]

result = alphabeta(
    0, 0, True, values,
    float('-inf'), float('inf')
)

print("Optimal value:", result)

Experiment 5: Single Layer Perceptron (AND Gate)

Aim: To implement a Single Layer Perceptron and train it to produce the AND gate output.

Python Code
# Experiment 5: Single Layer Perceptron
# Implementing AND Gate

# Training data: AND gate
X = [
    [0, 0],
    [0, 1],
    [1, 0],
    [1, 1]
]

Y = [0, 0, 0, 1]

# Initialize weights and bias
w1 = 0.0
w2 = 0.0
bias = 0.0

learning_rate = 0.1
epochs = 10

# Step activation function
def activation(value):
    if value >= 0:
        return 1
    return 0

# Training
for epoch in range(epochs):
    for i in range(len(X)):
        x1, x2 = X[i]
        target = Y[i]

        net = x1 * w1 + x2 * w2 + bias
        output = activation(net)

        error = target - output

        w1 = w1 + learning_rate * error * x1
        w2 = w2 + learning_rate * error * x2
        bias = bias + learning_rate * error

# Display results
print("Final weights:", w1, w2)
print("Final bias:", bias)

print("\nAND Gate Output:")
for inputs in X:
    net = inputs[0] * w1 + inputs[1] * w2 + bias
    output = activation(net)
    print(inputs, "->", output)

Experiment 6: Multi-Layer Perceptron using Supervised Learning

Aim: To demonstrate a Multi-Layer Perceptron (MLP) using supervised learning for the XOR problem.

Python Code
import math

# Sigmoid activation function
def sigmoid(x):
    return 1 / (1 + math.exp(-x))

# XOR input and target output
X = [[0, 0], [0, 1], [1, 0], [1, 1]]
Y = [0, 1, 1, 0]

# Fixed weights for hidden layer
W1 = [[20, 20], [-20, -20]]
B1 = [-10, 30]

# Fixed weights for output layer
W2 = [20, 20]
B2 = -30

print("Multi-Layer Perceptron using Supervised Learning")
print("Input\tPredicted Output")

for inputs in X:
    hidden = []
    for i in range(2):
        z = W1[i][0] * inputs[0] + W1[i][1] * inputs[1] + B1[i]
        hidden.append(sigmoid(z))

    output = sigmoid(W2[0] * hidden[0] + W2[1] * hidden[1] + B2)
    prediction = 1 if output >= 0.5 else 0

    print(inputs, "\t", prediction)

Experiment 7A: Self-Organizing Map (SOM)

Aim: To implement a Self-Organizing Map (SOM) for unsupervised learning and organize input patterns.

Python Code
import math

# Input data
data = [[1, 1], [1, 0], [0, 1], [0, 0]]

# Initial neuron weights
weights = [[0.2, 0.8], [0.8, 0.2]]

learning_rate = 0.5

# Train SOM
for inputs in data:
    # Find the Best Matching Unit (BMU)
    distances = []
    for w in weights:
        d = math.sqrt(
            (inputs[0] - w[0]) ** 2 +
            (inputs[1] - w[1]) ** 2
        )
        distances.append(d)

    winner = distances.index(min(distances))

    # Update winner's weights
    for j in range(2):
        weights[winner][j] += learning_rate * (
            inputs[j] - weights[winner][j]
        )

print("Final SOM Weights:")
for w in weights:
    print([round(v, 2) for v in w])

Experiment 7B: Learning Vector Quantization (LVQ)

Aim: To implement Learning Vector Quantization (LVQ) using labeled training data.

Python Code
import math

# Training data with class labels
data = [
    ([1, 1], 0),
    ([1, 0], 0),
    ([0, 1], 1),
    ([0, 0], 1)
]

# Initial prototype weights and their classes
weights = [[0.8, 0.8], [0.2, 0.2]]
labels = [0, 1]

learning_rate = 0.3

# Train LVQ
for inputs, target in data:
    distances = []
    for w in weights:
        d = math.sqrt(
            (inputs[0] - w[0]) ** 2 +
            (inputs[1] - w[1]) ** 2
        )
        distances.append(d)

    winner = distances.index(min(distances))

    # Move weight toward or away from input
    for j in range(2):
        if labels[winner] == target:
            weights[winner][j] += learning_rate * (
                inputs[j] - weights[winner][j]
            )
        else:
            weights[winner][j] -= learning_rate * (
                inputs[j] - weights[winner][j]
            )

print("Final LVQ Weights:")
for w in weights:
    print([round(v, 2) for v in w])

print("Prototype Classes:", labels)

Experiment 8A: Hopfield Network

Aim: To implement associative memory using a Hopfield Network and recall a stored pattern.

Python Code
pattern = [1, -1, 1]
n = len(pattern)

# Initialize weight matrix
weights = [[0] * n for _ in range(n)]

# Train the network
for i in range(n):
    for j in range(n):
        if i != j:
            weights[i][j] = pattern[i] * pattern[j]

# Noisy input
input_pattern = [1, -1, -1]
output = input_pattern.copy()

# Recall pattern
for i in range(n):
    total = sum(weights[i][j] * output[j] for j in range(n))
    output[i] = 1 if total >= 0 else -1

print("Stored Pattern:", pattern)
print("Input Pattern:", input_pattern)
print("Recalled Pattern:", output)

Aim: To implement BAM and demonstrate forward and backward recall between associated patterns.

Python Code
X = [1, -1]
Y = [1, -1, 1]

# Create weight matrix
weights = [
    [X[i] * Y[j] for j in range(len(Y))]
    for i in range(len(X))
]

print("Input X:", X)
print("Associated Y:", Y)

# Forward recall: X -> Y
output_y = [
    1 if sum(X[i] * weights[i][j] for i in range(len(X))) >= 0 else -1
    for j in range(len(Y))
]

print("Forward Recall:", output_y)

output_x = [
    1 if sum(Y[j] * weights[i][j] for j in range(len(Y))) >= 0 else -1
    for i in range(len(X))
]

print("Backward Recall:", output_x)

Experiment 9: Fuzzy Membership Functions and Fuzzification

Aim: To implement a triangular membership function and convert a crisp input into fuzzy membership values.

Python Code
# Triangular membership function
def triangular(x, a, b, c):
    if x <= a or x >= c:
        return 0.0
    elif x == b:
        return 1.0
    elif x < b:
        return (x - a) / (b - a)
    else:
        return (c - x) / (c - b)


# Crisp input
temperature = 30

# Membership values
cold = triangular(temperature, 0, 15, 30)
warm = triangular(temperature, 15, 30, 45)
hot = triangular(temperature, 30, 45, 60)

# Display results
print("Crisp Temperature:", temperature)
print("Fuzzy Membership Values:")
print("Cold:", round(cold, 2))
print("Warm:", round(warm, 2))
print("Hot:", round(hot, 2))
